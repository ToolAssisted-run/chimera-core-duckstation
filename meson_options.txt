option('minibox_dir', type: 'string', value: '', description: 'miniBox checkout (guest kit + host)')

option('recompiler', type: 'boolean', value: false,
  description: 'Build DuckStation\'s two x86-64 code generators beside its interpreters. They run without page faults (fastmem by lookup table, never by mmap); which executor a project uses is a setting.')

option('host_clock', type: 'boolean', value: false,
  description: 'DIAGNOSTIC, native builds only: leave DuckStation\'s timers on the host\'s clock, as upstream has them. The shipped core never does - its clock is the console\'s own, bought with executed instructions - and a reference built this way is how the gate\'s clock leg is seen to fail.')

option('mesa_guest_dir', type: 'string', value: '',
  description: 'A Mesa built for the guest (waterbox/setup-mesa.sh builds it into build/mesa). With one, this core also builds DuckStation\'s OpenGL device and its hardware renderer, and runs them against Mesa softpipe INSIDE the sandbox - a real GL implementation, no GPU, still deterministic. Without one, only the software renderer is built.')

option('gl_bridge', type: 'boolean', value: false,
  description: 'Native builds only: give run-wbx the host half of the GPU bridge, so the sandboxed hardware renderer can draw on a REAL graphics card (EGL, surfaceless). The GPU is outside the sandbox, outside the savestate and different on every machine; this is how the gate drives one, and what the frontend does through the engine\'s own host half.')

option('arcade', type: 'boolean', value: true,
  description: 'Build the arcade boards beside the console: ArcadeDuck\'s tree (extern/arcadeduck), compiled whole and linked in with its names kept to itself, so that one core.wbx is a PlayStation or one of the boards built out of one. Off, the core is the console alone.')
